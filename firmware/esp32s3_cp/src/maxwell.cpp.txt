#include <Arduino.h>
#include <SPI.h>

/* ================== User Wiring (ESP32-S3) ================== */
static const int PIN_SCK  = 7;
static const int PIN_MOSI = 15;
static const int PIN_MISO = 16;
static const int PIN_CS   = 13;
static const int PIN_RST  = 14;
static const int PIN_INT  = 42;

/* ================== Bus / Module Settings =================== */
static const uint8_t  MCP2515_CLK_MHZ     = 8;        // 8 MHz crystal
static const uint32_t CAN_BITRATE         = 125000;   // 125 kbps, extended
static const uint8_t  MAXWELL_MONITOR_ID  = 0x01;     // monitor addr
static const uint8_t  MAXWELL_GROUP_ADDR  = 0x01;     // group nibble
static uint8_t        MODULE_ADDR         = 0x01;     // your module id

/* ================= Limits / Policy ========================== */
static const uint32_t LV_MIN_mV = 200000, LV_MAX_mV = 500000;   // Low 200–500 V
static const uint32_t HV_MIN_mV = 400000, HV_MAX_mV = 1000000;  // High 400–1000 V
static const uint32_t HV_ENTER_mV = 500000;  // switch to High when >=500 V
static const uint32_t HV_EXIT_mV  = 400000;  // switch back to Low when <=400 V
static const uint32_t P_MAX_W     = 30000;   // 30 kW power cap

/* ================ Test Sweep Params (auto) ================== */
static const uint32_t CURRENT_LIMIT_mA     = 30000;   // ceiling, auto-reduced by Pmax/V
static const uint32_t START_VOLTAGE_V      = 50;      // will be clamped to >=200
static const uint32_t STOP_VOLTAGE_V       = 900;     // will be clamped to <=1000
static const uint32_t STEP_V               = 50;
static const uint32_t DWELL_MS             = 5000;

/* ================= MCP2515 SPI/Regs (subset) ================ */
static const uint8_t INSTR_RESET    = 0xC0;
static const uint8_t INSTR_READ     = 0x03;
static const uint8_t INSTR_WRITE    = 0x02;
static const uint8_t INSTR_BITMOD   = 0x05;
static const uint8_t INSTR_RTS      = 0x80;

static const uint8_t REG_CANSTAT    = 0x0E;
static const uint8_t REG_CANCTRL    = 0x0F;
static const uint8_t REG_CNF3       = 0x28;
static const uint8_t REG_CNF2       = 0x29;
static const uint8_t REG_CNF1       = 0x2A;
static const uint8_t REG_CANINTE    = 0x2B;
static const uint8_t REG_CANINTF    = 0x2C;
static const uint8_t REG_EFLG       = 0x2D;
static const uint8_t REG_TEC        = 0x1C;
static const uint8_t REG_REC        = 0x1D;

static const uint8_t REG_TXB0CTRL   = 0x30;
static const uint8_t REG_TXB0SIDH   = 0x31;
static const uint8_t REG_TXB0SIDL   = 0x32;
static const uint8_t REG_TXB0EID8   = 0x33;
static const uint8_t REG_TXB0EID0   = 0x34;
static const uint8_t REG_TXB0DLC    = 0x35;
static const uint8_t REG_TXB0D0     = 0x36;

static const uint8_t REQOP_MASK     = 0xE0;
static const uint8_t MODE_NORMAL    = 0x00;
static const uint8_t MODE_CONFIG    = 0x80;

/* ================= SPI Instance (ESP32-S3) ================== */
SPIClass spiFSPI(FSPI);
static inline void CS_LOW()  { digitalWrite(PIN_CS, LOW); }
static inline void CS_HIGH() { digitalWrite(PIN_CS, HIGH); }

/* ================= MCP2515 Low-level ======================== */
static void mcp_reset() { CS_LOW(); spiFSPI.transfer(INSTR_RESET); CS_HIGH(); delay(5); }
static uint8_t mcp_read(uint8_t a){ CS_LOW(); spiFSPI.transfer(INSTR_READ); spiFSPI.transfer(a); uint8_t v=spiFSPI.transfer(0); CS_HIGH(); return v; }
static void mcp_write(uint8_t a, uint8_t v){ CS_LOW(); spiFSPI.transfer(INSTR_WRITE); spiFSPI.transfer(a); spiFSPI.transfer(v); CS_HIGH(); }
static void mcp_writes(uint8_t a, const uint8_t* d, size_t n){ CS_LOW(); spiFSPI.transfer(INSTR_WRITE); spiFSPI.transfer(a); while(n--) spiFSPI.transfer(*d++); CS_HIGH(); }
static void mcp_bitmod(uint8_t a, uint8_t m, uint8_t d){ CS_LOW(); spiFSPI.transfer(INSTR_BITMOD); spiFSPI.transfer(a); spiFSPI.transfer(m); spiFSPI.transfer(d); CS_HIGH(); }

static bool mcp_setMode(uint8_t mode) {
  mcp_bitmod(REG_CANCTRL, REQOP_MASK, mode);
  for (uint32_t t=millis(); millis()-t<50; ) if ((mcp_read(REG_CANSTAT)&REQOP_MASK)==mode) return true;
  return false;
}

/* 125 kbps @ 8 MHz: 16 TQ, PropSeg=2, PS1=7, PS2=6, SJW=1, SAM=1 */
static bool mcp_setBitTiming_125k_8MHz() {
  mcp_write(REG_CNF1, 0x01);
  mcp_write(REG_CNF2, 0xF1);
  mcp_write(REG_CNF3, 0x05);
  return true;
}
static void mcp_acceptAll() {
  mcp_write(0x60, 0x64); // RXB0: any + BUKT
  mcp_write(0x70, 0x60); // RXB1: any
  mcp_write(REG_CANINTE, 0x03);
}

/* ================= Pack/Unpack 29-bit IDs =================== */
static void id29_to_regs(uint32_t id, uint8_t& sidh, uint8_t& sidl, uint8_t& eid8, uint8_t& eid0) {
  uint16_t sid = (id >> 18) & 0x7FF;
  uint32_t eid = id & 0x3FFFF;
  sidh = (sid >> 3) & 0xFF;
  sidl = ((sid & 0x7) << 5) | (1<<3) | ((eid >> 16) & 0x3);
  eid8 = (eid >> 8) & 0xFF;
  eid0 = eid & 0xFF;
}
static uint32_t regs_to_id29(uint8_t sidh, uint8_t sidl, uint8_t eid8, uint8_t eid0) {
  uint16_t sid = ((uint16_t)sidh<<3) | ((sidl>>5)&0x7);
  uint32_t eid = ((uint32_t)(sidl&0x3)<<16) | ((uint32_t)eid8<<8) | eid0;
  return ((uint32_t)sid<<18) | eid;
}

/* ================= TX/RX (extended) ========================= */
static bool mcp_send_ext(uint32_t id, const uint8_t* data, uint8_t len) {
  if (len > 8) len = 8;
  for (uint32_t t=millis(); millis()-t<50; ) if ((mcp_read(REG_TXB0CTRL)&0x08)==0) break;
  if (mcp_read(REG_TXB0CTRL)&0x08) return false;
  uint8_t sidh,sidl,eid8,eid0; id29_to_regs(id,sidh,sidl,eid8,eid0);
  uint8_t hdr[5] = { sidh,sidl,eid8,eid0,(uint8_t)(len&0x0F) };
  mcp_writes(REG_TXB0SIDH, hdr, 5);
  if (len) mcp_writes(REG_TXB0D0, data, len);
  CS_LOW(); spiFSPI.transfer(INSTR_RTS | 0x01); CS_HIGH();
  return true;
}
static bool mcp_receive_ext(uint32_t& id, uint8_t* data, uint8_t& len) {
  uint8_t intf = mcp_read(REG_CANINTF);
  if (intf & 0x01) { // RXB0
    uint8_t b[13]; CS_LOW(); spiFSPI.transfer(INSTR_READ); spiFSPI.transfer(0x61);
    for (int i=0;i<13;i++) b[i]=spiFSPI.transfer(0); CS_HIGH();
    id = regs_to_id29(b[0],b[1],b[2],b[3]); len=b[4]&0x0F; if(len>8) len=8; for(int i=5,j=0;j<len;i++,j++) data[j]=b[i];
    mcp_bitmod(REG_CANINTF, 0x01, 0x00); return true;
  }
  if (intf & 0x02) { // RXB1
    uint8_t b[13]; CS_LOW(); spiFSPI.transfer(INSTR_READ); spiFSPI.transfer(0x71);
    for (int i=0;i<13;i++) b[i]=spiFSPI.transfer(0); CS_HIGH();
    id = regs_to_id29(b[0],b[1],b[2],b[3]); len=b[4]&0x0F; if(len>8) len=8; for(int i=5,j=0;j<len;i++,j++) data[j]=b[i];
    mcp_bitmod(REG_CANINTF, 0x02, 0x00); return true;
  }
  return false;
}

/* ================= Diagnostics ============================== */
static void dump_errors(const char* tag){
  uint8_t eflg=mcp_read(REG_EFLG), tec=mcp_read(REG_TEC), rec=mcp_read(REG_REC), intf=mcp_read(REG_CANINTF);
  Serial.printf("[DIAG] %s EFLG=0x%02X TEC=%u REC=%u CANINTF=0x%02X\n", tag, eflg, tec, rec, intf);
}
static void dump_cnf() {
  Serial.printf("[CNF] CNF1=0x%02X CNF2=0x%02X CNF3=0x%02X (OSC=%u MHz, %lu bps)\n",
                mcp_read(REG_CNF1), mcp_read(REG_CNF2), mcp_read(REG_CNF3),
                MCP2515_CLK_MHZ, (unsigned long)CAN_BITRATE);
}

/* ================= Maxwell ENR Helpers ====================== */
namespace Maxwell {
enum MsgType : uint8_t { SetData=0x0, SetDataResp=0x1, ReadData=0x2, ReadDataResp=0x3 };
enum Cmd     : uint8_t { Vout_mV=0, Iout_mA=1, VoutRef_mV=2, IoutLimit_mA=3, PowerOnOff=4, ModuleStatus=8 };

static inline uint32_t buildId(uint8_t monitor, uint8_t module, uint8_t prodDay=0, uint16_t snLow=0) {
  return ((uint32_t)0x1<<25) | ((uint32_t)(monitor&0x0F)<<21) | ((uint32_t)(module&0x7F)<<14)
       | ((uint32_t)(prodDay&0x1F)<<9) | (uint32_t)(snLow&0x1FF);
}
static inline void put_be32(uint8_t *p, uint32_t v) { p[0]=v>>24; p[1]=v>>16; p[2]=v>>8; p[3]=v; }

static bool sendSet(uint8_t module, Cmd cmd, uint32_t value) {
  uint8_t d[8] = { (uint8_t)((MAXWELL_GROUP_ADDR<<4)|MsgType::SetData), (uint8_t)cmd, 0, 0, 0,0,0,0 };
  put_be32(&d[4], value);
  return mcp_send_ext(buildId(MAXWELL_MONITOR_ID, module), d, 8);
}
static bool sendRead(uint8_t module, Cmd cmd) {
  uint8_t d[8] = { (uint8_t)((MAXWELL_GROUP_ADDR<<4)|MsgType::ReadData), (uint8_t)cmd, 0, 0, 0,0,0,0 };
  return mcp_send_ext(buildId(MAXWELL_MONITOR_ID, module), d, 8);
}
static bool waitResp(uint8_t module, MsgType expectMsg, Cmd expectCmd, uint32_t &value, uint32_t timeoutMs=800) {
  uint8_t data[8], len=0; uint32_t id=0; uint32_t t0=millis();
  while (millis()-t0 < timeoutMs) {
    if (mcp_receive_ext(id, data, len)) {
      uint8_t mod = (id >> 14) & 0x7F;
      uint8_t msg = data[0] & 0x0F;
      uint8_t cmd = data[1];
      if (mod==module && msg==(uint8_t)expectMsg && cmd==(uint8_t)expectCmd) {
        value = (len>=8) ? ((uint32_t)data[4]<<24 | (uint32_t)data[5]<<16 | (uint32_t)data[6]<<8 | data[7]) : 0;
        return true;
      }
    }
    delay(1);
  }
  return false;
}

/* Public API */
static bool powerOn(uint8_t module, bool on=true) {
  if (!sendSet(module, PowerOnOff, on ? 0 : 1)) return false;
  uint32_t dummy=0; return waitResp(module, SetDataResp, PowerOnOff, dummy, 1200);
}
static bool setVref_mV(uint8_t module, uint32_t mV) {
  if (!sendSet(module, VoutRef_mV, mV)) return false;
  uint32_t echo=0; return waitResp(module, SetDataResp, VoutRef_mV, echo, 800);
}
static bool setILimit_mA(uint8_t module, uint32_t mA) {
  if (!sendSet(module, IoutLimit_mA, mA)) return false;
  uint32_t echo=0; return waitResp(module, SetDataResp, IoutLimit_mA, echo, 800);
}
static bool readVout_mV(uint8_t module, uint32_t &mV) {
  if (!sendRead(module, Vout_mV)) return false;
  return waitResp(module, ReadDataResp, Vout_mV, mV, 800);
}
static bool readIout_mA(uint8_t module, uint32_t &mA) {
  if (!sendRead(module, Iout_mA)) return false;
  return waitResp(module, ReadDataResp, Iout_mA, mA, 800);
}

/* ==== Hi/Lo mode control ==== */
enum HiLoMode : uint8_t { HILO_HIGH = 1, HILO_LOW = 2, HILO_AUTO = 3 };

// 0x5F: Set Hi/Lo/Auto; succeeds in shutdown
static bool setHiLoMode(uint8_t module, HiLoMode mode) {
  uint8_t d[8] = { (uint8_t)((MAXWELL_GROUP_ADDR<<4)|MsgType::SetData), 0x5F, 0,0, 0,0,0,0 };
  put_be32(&d[4], (uint32_t)mode);
  if (!mcp_send_ext(buildId(MAXWELL_MONITOR_ID, module), d, 8)) return false;
  uint32_t echo=0;
  return waitResp(module, SetDataResp, (Cmd)0x5F, echo, 1200);
}
// 0x60: Read configured Hi/Lo/Auto
static bool readHiLoModeCfg(uint8_t module, HiLoMode &mode) {
  if (!sendRead(module, (Cmd)0x60)) return false;
  uint32_t v=0; if (!waitResp(module, ReadDataResp, (Cmd)0x60, v, 800)) return false;
  mode = (HiLoMode)(v & 0xFF); return true;
}
// 0x65: Read *actual* active mode
static bool readHiLoModeActual(uint8_t module, HiLoMode &mode) {
  if (!sendRead(module, (Cmd)0x65)) return false;
  uint32_t v=0; if (!waitResp(module, ReadDataResp, (Cmd)0x65, v, 1200)) return false;
  mode = (HiLoMode)(v & 0xFF); return true;
}

} // namespace Maxwell

/* ================= Policy & Helpers ========================= */
static inline uint32_t clampV_mV(uint32_t v_mV){
  if (v_mV < LV_MIN_mV) return LV_MIN_mV;
  if (v_mV > HV_MAX_mV) return HV_MAX_mV;
  return v_mV;
}
static inline uint32_t pLimitedI_mA(uint32_t v_mV, uint32_t req_mA){
  if (v_mV < 1000) return 0;
  uint32_t ipow_mA = (uint32_t)((P_MAX_W * 1000ULL) / v_mV); // A = W/V ; mA = 1000*A
  return (req_mA < ipow_mA) ? req_mA : ipow_mA;
}

static bool ensureModeForTargetV(uint8_t module, uint32_t target_mV, Maxwell::HiLoMode &actual_out) {
  Maxwell::HiLoMode actual = Maxwell::HILO_LOW;
  // Read actual; fallback to configured if needed
  if (!Maxwell::readHiLoModeActual(module, actual)) {
    if (!Maxwell::readHiLoModeCfg(module, actual)) actual = Maxwell::HILO_LOW;
  }
  Maxwell::HiLoMode want = actual;
  if (actual == Maxwell::HILO_LOW  && target_mV >= HV_ENTER_mV) want = Maxwell::HILO_HIGH;
  if (actual == Maxwell::HILO_HIGH && target_mV <= HV_EXIT_mV)  want = Maxwell::HILO_LOW;

  if (want == actual) { actual_out = actual; return true; }

  Serial.printf("[MODE] Switching %s -> %s\n",
    actual==Maxwell::HILO_HIGH?"HIGH":"LOW", want==Maxwell::HILO_HIGH?"HIGH":"LOW");

  // Spec: 0x5F succeeds in shutdown
  (void)Maxwell::powerOn(module, false); delay(200);
  if (!Maxwell::setHiLoMode(module, want)) { actual_out = actual; return false; }
  (void)Maxwell::powerOn(module, true);

  // Confirm actual
  uint32_t t0 = millis(); Maxwell::HiLoMode now=actual;
  while (millis()-t0 < 2000) {
    if (Maxwell::readHiLoModeActual(module, now) && now == want) { actual_out = now; return true; }
    delay(100);
  }
  actual_out = now; return (now == want);
}

/* ================= Bring-up ================================ */
static bool mcp_begin() {
  pinMode(PIN_CS, OUTPUT);  CS_HIGH();
  pinMode(PIN_RST, OUTPUT); digitalWrite(PIN_RST, HIGH);
  pinMode(PIN_INT, INPUT_PULLUP);

  spiFSPI.begin(PIN_SCK, PIN_MISO, PIN_MOSI, PIN_CS);
  spiFSPI.beginTransaction(SPISettings(1000000, MSBFIRST, SPI_MODE0));

  digitalWrite(PIN_RST, LOW);  delay(2);
  digitalWrite(PIN_RST, HIGH); delay(10);

  mcp_reset();
  if (!mcp_setMode(MODE_CONFIG)) { Serial.println("[MCP] Cannot enter CONFIG"); return false; }
  if (!mcp_setBitTiming_125k_8MHz()) { Serial.println("[MCP] Bit timing set failed"); return false; }
  mcp_acceptAll();
  mcp_bitmod(REG_EFLG, 0xFF, 0x00);
  if (!mcp_setMode(MODE_NORMAL)) { Serial.println("[MCP] Cannot enter NORMAL"); return false; }

  Serial.println("[MCP] Ready @125 kbps (OSC=8 MHz, SAM=1)"); dump_cnf();
  return true;
}

/* =============== Automatic set (mode + Ilim + Vref) ========= */
static void set_voltage_auto(uint32_t target_V, uint32_t req_I_mA){
  // Clamp to module global range
  uint32_t target_mV = clampV_mV(target_V * 1000UL);

  // Ensure proper Hi/Lo for this voltage (with hysteresis)
  Maxwell::HiLoMode active{};
  if (!ensureModeForTargetV(MODULE_ADDR, target_mV, active)) {
    Serial.println("[MODE] ensureModeForTargetV failed (continuing best-effort)");
  } else {
    Serial.printf("[MODE] Active=%s\n", active==Maxwell::HILO_HIGH?"HIGH":
                               active==Maxwell::HILO_LOW ?"LOW":"AUTO");
  }

  // Dynamic Ilim based on power cap
  uint32_t ilim_mA = pLimitedI_mA(target_mV, req_I_mA);
  if (!Maxwell::setILimit_mA(MODULE_ADDR, ilim_mA)) {
    Serial.println("[SET] Ilim no-ack"); dump_errors("Ilimit");
  }

  // Finally set Vref
  Serial.printf("[SET] Vref=%lu V, Ilim=%.3f A\n", (unsigned long)(target_mV/1000), ilim_mA/1000.0f);
  if (!Maxwell::setVref_mV(MODULE_ADDR, target_mV)) {
    Serial.println("[SET] Vref set no-ack"); dump_errors("Vref");
  }
}

/* ================== Test Sweep ============================== */
static void voltage_sweep() {
  Serial.println("[TEST] Power ON module …");
  if (!Maxwell::powerOn(MODULE_ADDR,true)) { Serial.println("[TEST] PowerOn no-ack"); dump_errors("PowerOn"); }

  auto step = [](uint32_t vV){
    // One step: automatic mode + dynamic Ilim + set Vref + validate
    if (vV < 200) vV = 200; if (vV > 1000) vV = 1000;
    set_voltage_auto(vV, CURRENT_LIMIT_mA);
    delay(DWELL_MS);
    uint32_t vmeas=0;
    if (Maxwell::readVout_mV(MODULE_ADDR, vmeas))
      Serial.printf("[TEST] Read Vout = %.3f V\n", vmeas/1000.0f);
    else { Serial.println("[TEST] Read Vout timeout"); dump_errors("ReadV"); }
  };

  // Sweep up
  for (uint32_t v = START_VOLTAGE_V; v <= STOP_VOLTAGE_V; v += STEP_V) step(v);
  // Sweep down
  for (int32_t v = (int32_t)STOP_VOLTAGE_V - STEP_V; v >= 200; v -= STEP_V) step((uint32_t)v);

  Serial.println("[TEST] Setting Vref=0 and shutting down …");
  (void)Maxwell::setVref_mV(MODULE_ADDR, 0);
  delay(300);
  (void)Maxwell::powerOn(MODULE_ADDR, false);
  Serial.println("[TEST] Sweep complete");
}

/* ================= Arduino Hooks ============================ */
void setup() {
  Serial.begin(115200); delay(250);
  Serial.println("\n[BOOT] ESP32-S3 + MCP2515 (8 MHz) → Maxwell ENR (ID=0x01)");
  if (!mcp_begin()) { Serial.println("[BOOT] MCP init failed"); while(true) delay(1000); }

  // Optional: quick discover ping (not required for control)
  // (send a read and print any reply for 800 ms)
  uint32_t t0=millis(); uint8_t d8[8],len; uint32_t rid;
  Maxwell::sendRead(MODULE_ADDR, Maxwell::ModuleStatus);
  while (millis()-t0 < 800) if (mcp_receive_ext(rid,d8,len)) {
    Serial.printf("[DISC] got id=0x%08lX len=%u\n",(unsigned long)rid,len);
  }

  voltage_sweep();
}

void loop() {
  // Periodic monitor (not required for auto mode selection)
  static uint32_t t=0;
  if (millis()-t > 2000) {
    t = millis();
    uint32_t v=0,i=0;
    if (Maxwell::readVout_mV(MODULE_ADDR, v)) Serial.printf("[MON] Vout=%.3f V ", v/1000.0f); else Serial.print("[MON] Vout=? ");
    if (Maxwell::readIout_mA(MODULE_ADDR, i)) Serial.printf("Iout=%.3f A\n", i/1000.0f); else Serial.println("Iout=?");
  }
}
